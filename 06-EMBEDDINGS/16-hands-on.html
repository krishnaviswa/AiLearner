DO — cosine on toy vectors

No live embed API. Prove the geometry.

import math

def cosine(a, b):
    dot = sum(x * y for x, y in zip(a, b))
    na = math.sqrt(sum(x * x for x in a))
    nb = math.sqrt(sum(y * y for y in b))
    return dot / (na * nb)

q = [1.0, 0.0]
print(cosine(q, [1.0, 0.1]))
print(cosine(q, [0.0, 1.0]))
